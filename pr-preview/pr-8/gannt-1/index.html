"""Field-season Gantt chart with plotly.express.timeline.

Tasks are given as (label, row, start_day, duration_days) relative to day 0,
so re-planning only means editing the TASKS list or START_DATE.
"""
from datetime import date, timedelta

import pandas as pd
import plotly.express as px
TITLE = "Season 1, 27/28"

START_DATE = date(2027, 1, 4)   # placeholder: day 0 of the season
N_WEEKS = 4

# label, row (tasks sharing a row sit on one line), start day, duration (days)
TASKS = [
    ("Input from UG",              "Logistics",  0,  3),
    ("Establish camp",             "Logistics",  3,  3),
    ("Decamp / <br> depot kit",    "Logistics", 22,  3),
    ("Uplift to UG",               "Logistics", 25,  3),
    ("Setup RAID",                 "RAID setup", 5,  3),
    ("Drill RAID hole (O2)",       "Drilling",   8, 12),
    ("Image hole (O4)",       "Borehole instrumentation",   20,  2),
    ("Install<br>DTS (O5)",       "Borehole instrumentation",   22,  2),
    ("ApRES <br> survey (O3)",     "Geophysics", 7,  3),
    ("Active seismic survey (O3)", "Geophysics",10, 10),
]

ROW_ORDER = ["Logistics", "RAID setup", "Drilling", "Borehole instrumentation", "Geophysics"]

BAR = "#6b8aa3"
EDGE = "#1b2b3a"
HEADER = "#2e5f80"

df = pd.DataFrame(TASKS, columns=["Task", "Row", "StartDay", "Days"])
df["Start"] = pd.to_datetime([START_DATE + timedelta(days=d) for d in df.StartDay])
df["Finish"] = df.Start + pd.to_timedelta(df.Days, unit="D")

fig = px.timeline(
    df, x_start="Start", x_end="Finish", y="Row", text="Task",
    category_orders={"Row": ROW_ORDER},
)
fig.update_traces(
    marker=dict(color=BAR, line=dict(color=EDGE, width=2)),
    textposition="inside", insidetextanchor="middle",
    textfont=dict(color="white", size=13),
    hovertemplate="<b>%{text}</b><br>%{base|%d %b} – %{x|%d %b}<extra></extra>",
)

season_end = pd.Timestamp(START_DATE + timedelta(weeks=N_WEEKS))

# Alternating row bands
for i, _ in enumerate(ROW_ORDER):
    fig.add_hrect(y0=i - 0.5, y1=i + 0.5, layer="below", line_width=1,
                  fillcolor="#d5dbe0" if i % 2 == 0 else "#e8ecef")

# Week headers along the top
for w in range(N_WEEKS):
    x0 = pd.Timestamp(START_DATE + timedelta(weeks=w))
    x1 = x0 + pd.Timedelta(weeks=1)
    fig.add_shape(type="rect", xref="x", yref="paper", x0=x0, x1=x1,
                  y0=1.0, y1=1.20, fillcolor=HEADER,
                  line=dict(color="black", width=2))
    fig.add_annotation(xref="x", yref="paper", x=x0 + (x1 - x0) / 2, y=1.14,
                       text=f"Week {w + 1}", showarrow=False,
                       font=dict(color="white", size=14))

fig.update_xaxes(
    range=[pd.Timestamp(START_DATE), season_end],
    dtick=86400000.0,             # one gridline per day
    showticklabels=False,         # hide the day numbers
    showgrid=True, gridcolor="black", gridwidth=1,
    title=None,
)
fig.update_yaxes(title=None, showgrid=False)
fig.update_yaxes(title=None, showgrid=False, ticklabelstandoff=5)
fig.update_layout(
    title=dict(text=TITLE, x=0.55, xanchor="center", y=1, font=dict(size=22)),
    plot_bgcolor="white", paper_bgcolor="white",
    height=300, width=1200,
    margin=dict(l=110, r=20, t=70, b=10),
    showlegend=False,
    font=dict(family="Arial"),
    bargap=0,          # bars fill the full height of each row band
)

if __name__ == "__main__":
    fig.write_html("season1_gantt.html")
    # Static export needs `pip install kaleido`
    try:
        fig.write_image("season1_gantt.png", scale=2)
    except Exception as e:
        print("PNG export skipped:", e)
    fig.show()
